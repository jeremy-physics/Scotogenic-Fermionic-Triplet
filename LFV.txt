m = {meexp, m\[Mu]exp, m\[Tau]exp};

F[x_] := (1 - 6 x + 3 x^2 + 2 x^3 - 6 x^2 Log[x])/(6 (1 - x)^4); 

G[x_] := (2 - 9 x + 18 x^2 - 11 x^3 + 6 x^3 Log[x])/(6 (1 - x)^4);

D1[x_, y_] := -(1/((1 - x) (1 - y))) - (
   x^2 Log[x])/((1 - x)^2 (x - y)) - (y^2 Log[x])/((1 - y)^2 (y - x));

D2[x_, y_] := -(1/((1 - x) (1 - y))) - (
   x Log[x])/((1 - x)^2 (x - y)) - (y Log[x])/((1 - y)^2 (y - x));

D1p[x_] := (-1 + x^2 - 2 x Log[x])/(1 - x)^3;

D2p[x_] := (-2 + 2 x - (1 + x) Log[x])/(1 - x)^3;

\[Xi] = {\[Xi]1, \[Xi]2};
m\[Eta] = {m\[Eta]1bp, m\[Eta]2bp};

AND[i_, j_, xN_] := 
  1/(6 (4 \[Pi])^2)
    Sum[1/m\[Eta][[
      k]]^2 (xN[[i]][[s]] xN[[j]][[s]] G[\[Xi][[k]]]), {s, 2}, {k, 
     2}];(*i->j*)

AD[i_, j_, xN_] := 
  1/(2 (4 \[Pi])^2)
    Sum[1/m\[Eta][[
      k]]^2 (xN[[i]][[s]] xN[[j]][[s]] F[\[Xi][[k]]]), {s, 2}, {k, 2}];

B[\[Alpha]_, \[Beta]_, xN_] := 
  1/(((4 \[Pi])^2) 4 \[Pi] \[Alpha]em )
    Sum[1/
     m\[Eta][[
      i]]^2 (1/
        2 D1p[\[Xi][[i]]] Conjugate[
         xN[[\[Beta]]][[i]]] xN[[\[Beta]]][[i]] Conjugate[
         xN[[\[Beta]]][[i]]] xN[[\[Alpha]]][[i]] + 
       Sqrt[\[Xi][[i]] \[Xi][[i]]]
         D2p[\[Xi][[i]]] Conjugate[xN[[\[Beta]]][[i]]] Conjugate[
         xN[[\[Beta]]][[i]]]  xN[[\[Beta]]][[i]] xN[[\[Alpha]]][[
         i]]), {i, 2}];

Br[i_, j_, xN_] := (3 (4 Pi)^3 \[Alpha]em)/(4 GF^2)
    Abs[AD[i, j, xN]]^2;

Br2[i_, j_, xN_] := (3 (4 \[Pi])^2 \[Alpha]em^2)/(
   8 GF^2) (Abs[AND[i, j, xN]]^2 + 
     Abs[AD[i, j, xN]]^2 (16/3 Log[m[[i]]/m[[j]]] - 22/3) + 
     1/6 Abs[B[i, j, 
        xN]]^2 + (-2 AND[i, j, xN] Conjugate[AD[i, j, xN]] + 
       1/3 AND[i, j, xN] Conjugate[B[i, j, xN]] - 
       2/3 AD[i, j, xN] Conjugate[B[i, j, xN]] + 
       Conjugate[-2 AND[i, j, xN] Conjugate[AD[i, j, xN]] + 
         1/3 AND[i, j, xN] Conjugate[B[i, j, xN]] - 
         2/3 AD[i, j, xN] Conjugate[B[i, j, xN]]]));

 Qu = 2/3; Qd = -(1/3); Qs = -(1/3);

rule = {pe -> m\[Mu]exp, Ee -> m\[Mu]exp};

g\[Gamma]LVu[i_, j_, xN_] := 
 Sqrt[2]/GF 4 \[Pi] \[Alpha]em Qu (AND[i, j, xN] - AD[i, j, xN]); 
g\[Gamma]LVd[i_, j_, xN_] := 
 Sqrt[2]/GF 4 \[Pi] \[Alpha]em Qd (AND[i, j, xN] - AD[i, j, xN]); 
g\[Gamma]LVs[i_, j_, xN_] := 
 Sqrt[2]/GF 4 \[Pi] \[Alpha]em Qs (AND[i, j, xN] - AD[i, j, xN]);

gLVu[i_, j_, xN_] := g\[Gamma]LVu[i, j, xN]; 
gLVd[i_, j_, xN_] := g\[Gamma]LVd[i, j, xN]; 
gLVs[i_, j_, xN_] := g\[Gamma]LVs[i, j, xN];

gRVu[i_, j_, xN_] := g\[Gamma]LVu[i, j, xN]; 
gRVd[i_, j_, xN_] := g\[Gamma]LVd[i, j, xN]; 
gRVs[i_, j_, xN_] := g\[Gamma]LVs[i, j, xN];

gLSu = 0; gLSd = 0; gLSs = 0;

gRSu = 0; gRSd = 0; gRSs = 0;

GVup = 2; GVdn = 2; GVdp = 1; GVun = 1; GSup = 5.1; GSdn = 5.1; GSdp \
= 4.3; GSun = 4.3; GSsp = 2.5; GSsn = 2.5; GVsp = 0; GVsn = 0;

GVu = {GVup, GVun};
GVd = {GVdp, GVdn};
GVs = {GVsp, GVsn};
GSu = {GSup, GSun};
GSd = {GSdp, GSdn};
GSs = {GSsp, GSsn};

gLV[i_, j_, xN_] := {gLVu[i, j, xN], gLVd[i, j, xN], gLVs[i, j, xN]};
gRV[i_, j_, xN_] := {gRVu[i, j, xN], gRVd[i, j, xN], gRVs[i, j, xN]};
gLS[i_, j_, xN_] := {gLSu, gLSd, gLSs};
gRS[i_, j_, xN_] := {gRSu, gRSd, gRSs};

g0LV[i_, j_, xN_] := 
  1/2 (gLV[i, j, xN][[1]] GVu[[1]] + gLV[i, j, xN][[2]] GVd[[1]] + 
     gLV[i, j, xN][[3]] GVs[[1]] + gLV[i, j, xN][[1]] GVu[[2]] + 
     gLV[i, j, xN][[2]] GVd[[2]] + gLV[i, j, xN][[3]] GVs[[2]]);
g1LV[i_, j_, xN_] := 
  1/2 (gLV[i, j, xN][[1]] GVu[[1]] + gLV[i, j, xN][[2]] GVd[[1]] + 
     gLV[i, j, xN][[3]] GVs[[1]] - gLV[i, j, xN][[1]] GVu[[2]] - 
     gLV[i, j, xN][[2]] GVd[[2]] - gLV[i, j, xN][[3]] GVs[[2]]);
g0RV[i_, j_, xN_] := 
  1/2 (gRV[i, j, xN][[1]] GVu[[1]] + gRV[i, j, xN][[2]] GVd[[1]] + 
     gRV[i, j, xN][[3]] GVs[[1]] + gRV[i, j, xN][[1]] GVu[[2]] + 
     gRV[i, j, xN][[2]] GVd[[2]] + gRV[i, j, xN][[3]] GVs[[2]]);
g1RV[i_, j_, xN_] := 
  1/2 (gRV[i, j, xN][[1]] GVu[[1]] + gRV[i, j, xN][[2]] GVd[[1]] + 
     gRV[i, j, xN][[3]] GVs[[1]] - gRV[i, j, xN][[1]] GVu[[2]] - 
     gRV[i, j, xN][[2]] GVd[[2]] - gRV[i, j, xN][[3]] GVs[[2]]);
g0LS[i_, j_, xN_] := 
  1/2 (gLS[i, j, xN][[1]] GSu[[1]] + gLS[i, j, xN][[2]] GSd[[1]] + 
     gLS[i, j, xN][[3]] GSs[[1]] + gLS[i, j, xN][[1]] GSu[[2]] + 
     gLS[i, j, xN][[2]] GVd[[2]] + gLS[i, j, xN][[3]] GSs[[2]]);
g1LS[i_, j_, xN_] := 
  1/2 (gLS[i, j, xN][[1]] GSu[[1]] + gLS[i, j, xN][[2]] GSd[[1]] + 
     gLS[i, j, xN][[3]] GSs[[1]] - gLS[i, j, xN][[1]] GSu[[2]] - 
     gLS[i, j, xN][[2]] GVd[[2]] - gLS[i, j, xN][[3]] GSs[[2]]);
g0RS[i_, j_, xN_] := 
  1/2 (gRS[i, j, xN][[1]] GSu[[1]] + gRS[i, j, xN][[2]] GSd[[1]] + 
     gRS[i, j, xN][[3]] GSs[[1]] + gRS[i, j, xN][[1]] GSu[[2]] + 
     gRS[i, j, xN][[2]] GVd[[2]] + gRS[i, j, xN][[3]] GSs[[2]]);
g1RS[i_, j_, xN_] := 
  1/2 (gRS[i, j, xN][[1]] GSu[[1]] + gRS[i, j, xN][[2]] GSd[[1]] + 
     gRS[i, j, xN][[3]] GSs[[1]] - gRS[i, j, xN][[1]] GSu[[2]] - 
     gRS[i, j, xN][[2]] GVd[[2]] - gRS[i, j, xN][[3]] GSs[[2]]);

CR[Z_, n_, Zeff_, Fp_, \[CapitalGamma]c_, i_, j_, xN_] := (
   pe Ee m\[Mu]exp^3 GF^2 \[Alpha]em^3 Zeff^4 Fp^2)/(
   8 \[Pi]^2 Z \[CapitalGamma]c) (Abs[(Z + n) (g0LV[i, j, xN] + 
          g0LS[i, j, xN]) + (Z - n) (g1LV[i, j, xN] + 
          g1LS[i, j, xN])]^2 + 
     Abs[(Z + n) (g0RV[i, j, xN] + g0RS[i, j, xN]) + (Z - 
          n) (g1RV[i, j, xN] + g1RS[i, j, xN])]^2);
